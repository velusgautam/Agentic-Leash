# Milestone 1: App Shell Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A T3 Code-style Agentic Leash window with sidebar navigation and a settings skeleton. Closing it or pressing ⌘Q hides it to the menu bar, only the menu bar Quit exits, and the installed app starts hidden at login.

**Architecture:** electron-vite builds three bundles: main (lifecycle, tray, menus, login item), a sandboxed preload (a small `window.leash` bridge) and the React renderer. The renderer uses UI components and CSS copied from T3 Code at a pinned commit, TanStack Router with hash history, and Tailwind 4. Vitest covers the pure main-process modules; Playwright drives the built app for the lifecycle and navigation checks.

**Tech Stack:** Electron 44.7.2, electron-vite 6, Vite 8, React 19, TypeScript 5.9, Tailwind 4.3, `@base-ui/react` 1.x, `lucide-react` 0.564.0, `@tanstack/react-router` 1.x, Vitest 5, Playwright 1.64.

**Parent plan:** [`2026-10-15-agentic-leash-build-plan.md`](2026-10-15-agentic-leash-build-plan.md), Milestone 1.

## Global Constraints

- Every Global Constraint in the parent plan applies.
- Package manager: pnpm 12. `pnpm-workspace.yaml` sets `allowBuilds` (pnpm fails the install on unapproved build scripts).
- Renderer packages are `devDependencies`, because Vite bundles them. Only packages the main process loads at runtime go in `dependencies`. M1 has none.
- `BrowserWindow` uses `sandbox: true` and `contextIsolation: true`. The preload may only import `electron`.
- T3 Code source comes from commit `ede3f2d6ba4943ba0a14aa4af656cfa537a1efe6`. Copied files stay as close to upstream as possible; every edit is listed in its task.
- Import alias `~` maps to `src/renderer/src`, the same alias T3 Code uses, so copied files need no import rewrites.
- No ESLint or Prettier in M1. The gates are `pnpm typecheck`, `pnpm test` and `pnpm test:e2e`.
- `AGENTIC_LEASH_USER_DATA` (env) overrides Electron's `userData` folder. It's the only test hook; e2e runs use it for isolation.
- Window chrome copied from T3 Code: `titleBarStyle: "hiddenInset"`, traffic lights at `{ x: 16, y: 19 }`, 52 px top bar, `minWidth` 840, `minHeight` 620, background `#0a0a0a` dark / `#ffffff` light.

```ts
import { join } from "node:path";
import { app, BrowserWindow, nativeTheme, shell } from "electron";

if (process.env.AGENTIC_LEASH_USER_DATA) app.setPath("userData", process.env.AGENTIC_LEASH_USER_DATA);

function createWindow(visible: boolean): BrowserWindow {
  const window = new BrowserWindow({
    width: 1100,
    height: 720,
    minWidth: 840,
    minHeight: 620,
    show: false,
    title: "Agentic Leash",
    titleBarStyle: "hiddenInset",
    trafficLightPosition: { x: 16, y: 19 },
    backgroundColor: nativeTheme.shouldUseDarkColors ? "#0a0a0a" : "#ffffff",
    webPreferences: { preload: join(__dirname, "../preload/index.js"), sandbox: true },
  });
  if (visible) window.once("ready-to-show", () => window.show());
  window.webContents.setWindowOpenHandler(({ url }) => {
    void shell.openExternal(url);
    return { action: "deny" };
  });
  if (!app.isPackaged && process.env.ELECTRON_RENDERER_URL) void window.loadURL(process.env.ELECTRON_RENDERER_URL);
  else void window.loadFile(join(__dirname, "../renderer/index.html"));
  return window;
}

void app.whenReady().then(() => createWindow(true));
```

```ts
// src/main/index.ts
import { join } from "node:path";
import { app, BrowserWindow, Menu, Notification, nativeTheme, shell, Tray } from "electron";
import trayIcon from "../../resources/trayTemplate.png?asset";
import { appMenuTemplate, trayMenuTemplate } from "./menus";
import { readUiState, writeUiState } from "./uiState";

if (process.env.AGENTIC_LEASH_USER_DATA) app.setPath("userData", process.env.AGENTIC_LEASH_USER_DATA);

let win: BrowserWindow | null = null;
// Module scope on purpose: a garbage-collected Tray vanishes from the menu bar.
let tray: Tray | null = null;
let quitting = false;

function createWindow(visible: boolean): BrowserWindow {
  const window = new BrowserWindow({
    width: 1100,
    height: 720,
    minWidth: 840,
    minHeight: 620,
    show: false,
    title: "Agentic Leash",
    titleBarStyle: "hiddenInset",
    trafficLightPosition: { x: 16, y: 19 },
    backgroundColor: nativeTheme.shouldUseDarkColors ? "#0a0a0a" : "#ffffff",
    webPreferences: { preload: join(__dirname, "../preload/index.js"), sandbox: true },
  });
  if (visible) window.once("ready-to-show", () => window.show());
  window.webContents.setWindowOpenHandler(({ url }) => {
    void shell.openExternal(url);
    return { action: "deny" };
  });
  if (!app.isPackaged && process.env.ELECTRON_RENDERER_URL) void window.loadURL(process.env.ELECTRON_RENDERER_URL);
  else void window.loadFile(join(__dirname, "../renderer/index.html"));
  return window;
}

function showWindow(): void {
  if (!win) return;
  void app.dock?.show();
  win.show();
  win.focus();
}

function hideToMenuBar(): void {
  if (!win) return;
  win.hide();
  app.dock?.hide();
  const dir = app.getPath("userData");
  const state = readUiState(dir);
  if (state.hideNoticeShown) return;
  new Notification({
    title: "Agentic Leash is still running",
    body: "It stays in the menu bar. Quit from the menu bar icon.",
  }).show();
  writeUiState(dir, { ...state, hideNoticeShown: true });
}

if (!app.requestSingleInstanceLock()) {
  app.quit();
} else {
  app.on("second-instance", showWindow);
  app.on("activate", showWindow);
  // Also fires for tray Quit, Dock > Quit, logout and shutdown; those must really quit.
  app.on("before-quit", () => {
    quitting = true;
  });

  void app.whenReady().then(() => {
    const hidden = process.argv.includes("--hidden");
    win = createWindow(!hidden);
    win.on("close", (event) => {
      if (quitting) return;
      event.preventDefault();
      hideToMenuBar();
    });
    if (hidden) app.dock?.hide();
    Menu.setApplicationMenu(Menu.buildFromTemplate(appMenuTemplate({ hide: hideToMenuBar, dev: !app.isPackaged })));
    tray = new Tray(trayIcon);
    tray.setToolTip("Agentic Leash");
    tray.setContextMenu(Menu.buildFromTemplate(trayMenuTemplate({ open: showWindow, quit: () => app.quit() })));
  });
}
```

The local is `window`, not `win`, so it doesn't shadow the module-level `win`.

```ts
// src/main/loginItem.ts
import { existsSync, mkdirSync, rmSync, writeFileSync } from "node:fs";
import { homedir } from "node:os";
import { dirname, join } from "node:path";

// LaunchAgent, not app.setLoginItemSettings: login items silently fail for non-notarized apps.
export const LOGIN_AGENT_LABEL = "com.agenticleash.app";

const escapeXml = (s: string) => s.replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;");

export function loginAgentPath(home = homedir()): string {
  return join(home, "Library", "LaunchAgents", `${LOGIN_AGENT_LABEL}.plist`);
}

export function loginAgentPlist(appBundlePath: string): string {
  return `<?xml version="1.0" encoding="UTF-8"?>
<!DOCTYPE plist PUBLIC "-//Apple//DTD PLIST 1.0//EN" "http://www.apple.com/DTDs/PropertyList-1.0.dtd">
<plist version="1.0">
<dict>
  <key>Label</key><string>${LOGIN_AGENT_LABEL}</string>
  <key>ProgramArguments</key>
  <array>
    <string>/usr/bin/open</string>
    <string>-g</string>
    <string>-a</string>
    <string>${escapeXml(appBundlePath)}</string>
    <string>--args</string>
    <string>--hidden</string>
  </array>
  <key>RunAtLoad</key><true/>
  <key>LimitLoadToSessionType</key><string>Aqua</string>
</dict>
</plist>
`;
}

export function isOpenAtLogin(home?: string): boolean {
  return existsSync(loginAgentPath(home));
}

export function setOpenAtLogin(enabled: boolean, appBundlePath: string, home?: string): void {
  const path = loginAgentPath(home);
  if (!enabled) {
    rmSync(path, { force: true });
    return;
  }
  mkdirSync(dirname(path), { recursive: true });
  writeFileSync(path, loginAgentPlist(appBundlePath));
}
```

The plist is only written, never loaded with `launchctl`: loading it now would run `RunAtLoad` and start a second copy. It takes effect at the next login.

```ts
// src/main/index.ts
import { join, resolve } from "node:path";
import { app, BrowserWindow, ipcMain, Menu, Notification, nativeTheme, shell, Tray } from "electron";
import trayIcon from "../../resources/trayTemplate.png?asset";
import type { OpenAtLoginState } from "../shared/bridge";
import { isOpenAtLogin, setOpenAtLogin } from "./loginItem";
import { appMenuTemplate, trayMenuTemplate } from "./menus";
import { readUiState, writeUiState } from "./uiState";

if (process.env.AGENTIC_LEASH_USER_DATA) app.setPath("userData", process.env.AGENTIC_LEASH_USER_DATA);

let win: BrowserWindow | null = null;
// Module scope on purpose: a garbage-collected Tray vanishes from the menu bar.
let tray: Tray | null = null;
let quitting = false;

function createWindow(visible: boolean): BrowserWindow {
  const window = new BrowserWindow({
    width: 1100,
    height: 720,
    minWidth: 840,
    minHeight: 620,
    show: false,
    title: "Agentic Leash",
    titleBarStyle: "hiddenInset",
    trafficLightPosition: { x: 16, y: 19 },
    backgroundColor: nativeTheme.shouldUseDarkColors ? "#0a0a0a" : "#ffffff",
    webPreferences: { preload: join(__dirname, "../preload/index.js"), sandbox: true },
  });
  if (visible) window.once("ready-to-show", () => window.show());
  window.webContents.setWindowOpenHandler(({ url }) => {
    void shell.openExternal(url);
    return { action: "deny" };
  });
  if (!app.isPackaged && process.env.ELECTRON_RENDERER_URL) void window.loadURL(process.env.ELECTRON_RENDERER_URL);
  else void window.loadFile(join(__dirname, "../renderer/index.html"));
  return window;
}

function showWindow(): void {
  if (!win) return;
  void app.dock?.show();
  win.show();
  win.focus();
}

function hideToMenuBar(): void {
  if (!win) return;
  win.hide();
  app.dock?.hide();
  const dir = app.getPath("userData");
  const state = readUiState(dir);
  if (state.hideNoticeShown) return;
  new Notification({
    title: "Agentic Leash is still running",
    body: "It stays in the menu bar. Quit from the menu bar icon.",
  }).show();
  writeUiState(dir, { ...state, hideNoticeShown: true });
}

// <bundle>.app/Contents/MacOS/<exe> → <bundle>.app
const appBundlePath = () => resolve(process.execPath, "../../..");

function loginItemState(): OpenAtLoginState {
  return { available: app.isPackaged, enabled: app.isPackaged && isOpenAtLogin() };
}

function registerLoginItem(): void {
  ipcMain.handle("login-item:get", () => loginItemState());
  ipcMain.handle("login-item:set", (_event, enabled: boolean) => {
    if (app.isPackaged) setOpenAtLogin(enabled, appBundlePath());
    return loginItemState();
  });
  const dir = app.getPath("userData");
  const state = readUiState(dir);
  if (app.isPackaged && !state.loginItemConfigured) {
    setOpenAtLogin(true, appBundlePath());
    writeUiState(dir, { ...state, loginItemConfigured: true });
  }
}

if (!app.requestSingleInstanceLock()) {
  app.quit();
} else {
  app.on("second-instance", showWindow);
  app.on("activate", showWindow);
  // Also fires for tray Quit, Dock > Quit, logout and shutdown; those must really quit.
  app.on("before-quit", () => {
    quitting = true;
  });

  void app.whenReady().then(() => {
    registerLoginItem();
    const hidden = process.argv.includes("--hidden");
    win = createWindow(!hidden);
    win.on("close", (event) => {
      if (quitting) return;
      event.preventDefault();
      hideToMenuBar();
    });
    if (hidden) app.dock?.hide();
    Menu.setApplicationMenu(Menu.buildFromTemplate(appMenuTemplate({ hide: hideToMenuBar, dev: !app.isPackaged })));
    tray = new Tray(trayIcon);
    tray.setToolTip("Agentic Leash");
    tray.setContextMenu(Menu.buildFromTemplate(trayMenuTemplate({ open: showWindow, quit: () => app.quit() })));
  });
}
```
